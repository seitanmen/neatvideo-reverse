# Check build of the widget port (qtui/src/widgets) apart from the main qtui build.
cmake_minimum_required(VERSION 3.21)
project(widgets_check LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets)
set(Q ${CMAKE_CURRENT_SOURCE_DIR}/../../../../../qtui)
set(NV ${CMAKE_CURRENT_SOURCE_DIR}/../../../../../native)
add_compile_options(-ffp-contract=off -fno-fast-math -Wall -Wextra)
file(GLOB W ${Q}/src/widgets/*.cpp ${Q}/src/widgets/*.h)
add_executable(wcheck main.cpp ${W} ${Q}/src/core/pyobj.cpp ${Q}/src/core/state.cpp ${Q}/src/core/state.h)
target_include_directories(wcheck PRIVATE ${Q}/src/core ${Q}/src/widgets ${NV})
target_link_libraries(wcheck PRIVATE Qt6::Widgets ${NV}/libnvcore.dylib)
set_target_properties(wcheck PROPERTIES BUILD_RPATH "${NV}")
